"""
Find the symbolic skeleton of an ODE system that represents Acute Kidney Injury (AKI)
progression in middle-aged patients (50–64 years old) during the first 168 hours 
(7 days) after hospital admission. The system should incorporate biologically 
plausible interactions among the following biomarkers:

    - Creatinine (mg/dL): Indicator of renal filtration function
    - BUN – Blood Urea Nitrogen (mg/dL): Accumulates when renal clearance fails
    - Potassium (mmol/L): Reflects renal electrolyte regulation and impaired excretion

The time units is in hours.

HARD CONSTRAINT: The parameter array has exactly MAX_NPARAMS = 40 slots (indices 0–39).
You may use ONLY params[0] through params[39]. You must count every index you assign.
The total number of distinct indices used must be <= 40.
If you need a new parameter, you must REUSE or REMOVE an existing one.
Do NOT introduce any params[k] with k >= 40 under any circumstances.
"""

import torch

# Initialize parameters
MAX_NPARAMS = 40
params = torch.ones(MAX_NPARAMS)


@system.evolve
def system(biomarkers: torch.Tensor, params: torch.Tensor, t: float) -> torch.Tensor:
    """Symbolic skeleton for a 3-variable AKI ODE system.
    Args:
        biomarkers: A torch tensor containing
            [Creatinine, BUN, Potassium].
        params: Array of numeric constants or parameters to be optimized.
    Returns:
        A torch tensor representing the temporal derivatives (d/dt)
        of the biomarkers.
    """
    Cr = biomarkers[..., 0]
    BUN = biomarkers[..., 1]
    K = biomarkers[..., 2]

    # Renal recovery with injury feedback and time-dependent saturation
    R_min, R_max, k_rec = params[..., 0], params[..., 1], params[..., 2]
    inhib_Cr = torch.sigmoid(params[..., 19] * (Cr - params[..., 20]))
    inhib_BUN = torch.sigmoid(params[..., 21] * (BUN - params[..., 22]))
    recovery_rate = k_rec * (1.0 - 0.5 * (inhib_Cr + inhib_BUN))
    R_sat = params[..., 27]
    R_raw = R_min + (R_max - R_min) * (1.0 - torch.exp(-recovery_rate * t))
    R = R_raw * torch.exp(-t / R_sat) + R_max * (1.0 - torch.exp(-t / R_sat))
    R = torch.clamp(R, 0.0, 1.5)

    # Creatinine: saturated clearance + muscle breakdown feedback
    p_Cr, k_CrR, k_CrNR = params[..., 3], params[..., 4], params[..., 5]
    Cr_ref, sat_Cr = params[..., 6], params[..., 23]
    clearance_Cr = (k_CrNR + k_CrR * R) * (Cr - Cr_ref) / (1.0 + (Cr - Cr_ref) / sat_Cr)
    # Add muscle catabolism term driven by hyperkalemia
    cat_Cr = torch.clamp(K - params[..., 34], min=0.0) * params[..., 35]
    dCr = p_Cr + cat_Cr - clearance_Cr

    # BUN: catabolic boost from hyperkalemia and high Cr, clearance inhibition by hyperkalemia
    p_BUN_base, k_BUNR, k_BUNNR = params[..., 7], params[..., 8], params[..., 9]
    BUN_ref, K_cat_thresh = params[..., 10], params[..., 24]
    cat_boost_K = torch.clamp(K - K_cat_thresh, min=0.0) * params[..., 25]
    cat_boost_Cr = torch.clamp(Cr - params[..., 28], min=0.0) * params[..., 29]
    p_BUN = p_BUN_base + cat_boost_K + cat_boost_Cr
    K_inhib = torch.sigmoid(params[..., 30] * (K - params[..., 31]))
    effective_R_BUN = R * (1.0 - 0.3 * K_inhib)
    dBUN = p_BUN - (k_BUNNR + k_BUNR * effective_R_BUN) * (BUN - BUN_ref)

    # Potassium: saturated renal excretion, buffering impaired by uremia, cellular release
    K_ref, k_Kbuf, k_KR = params[..., 11], params[..., 12], params[..., 13]
    s_K0, s_Kcr, s_Kbun = params[..., 14], params[..., 15], params[..., 16]
    Cr_sev0, BUN_sev0 = params[..., 17], params[..., 18]
    sev = (torch.clamp(Cr - Cr_sev0, min=0.0) * s_Kcr +
           torch.clamp(BUN - BUN_sev0, min=0.0) * s_Kbun)
    K_exc_sat = params[..., 26]
    renal_exc = (k_KR * R) * (K - K_ref) / (1.0 + (K - K_ref) / K_exc_sat)
    buf_impair = torch.sigmoid(params[..., 32] * (BUN - params[..., 33]))
    effective_k_Kbuf = k_Kbuf * (1.0 - 0.4 * buf_impair)
    # Add acidosis-driven shift from hypercatabolism (Cr-linked)
    acid_shift = torch.clamp(Cr - params[..., 36], min=0.0) * params[..., 37]
    dK = (
        - effective_k_Kbuf * (K - K_ref)
        - renal_exc
        + (1.0 - torch.clamp(R, 0.0, 1.0)) * s_K0
        + (1.0 - torch.clamp(R, 0.0, 1.0)) * sev
        + acid_shift
    )

    d_biomarkers = torch.stack([dCr, dBUN, dK], dim=-1)
    return d_biomarkers
